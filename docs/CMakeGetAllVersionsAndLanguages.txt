# ============================================================
# 🌐 Gather All Documentation Versions and Languages
# ------------------------------------------------------------
# This script identifies the canonical source languages used by
# the versioned Sphinx output.
#
# It populates the following variables:
#   - ALL_VERSIONS : The current documentation version.
#   - ALL_LANGUAGES: List of all languages across versions.
# ============================================================


# ------------------------------------------------------------
# 🗂️ Set the current documentation version
# ------------------------------------------------------------
set(ALL_VERSIONS "${LATEST_VERSION}")


# ------------------------------------------------------------
# 🌍 Collect all languages for each version
# ------------------------------------------------------------
set(LATEST_VERSION_SOURCE_DIR "${WEAVER_DOCS_SOURCE_DIR}/${LATEST_VERSION}")

# Start with the default language
set(ALL_LANGUAGES "${DEFAULT_LANGUAGE}")

# Add languages present in the canonical source
file(GLOB latest_languages RELATIVE "${LATEST_VERSION_SOURCE_DIR}/locale" "${LATEST_VERSION_SOURCE_DIR}/locale/*")
if(latest_languages)
    list(APPEND ALL_LANGUAGES "${latest_languages}")
endif()

list(REMOVE_DUPLICATES ALL_LANGUAGES)
