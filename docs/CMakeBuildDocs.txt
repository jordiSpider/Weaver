include("${DOCS_DIR}/CMakeConfigureSphinxConf.txt")

set(SPHINX_DOCTREE_DIR "${SPHINX_BUILD_DIR}/doctrees")
set(DOCS_WEAVER_CATALOG "${SPHINX_DOCS_OUTPUT_DIR}/versions.json")

if(TARGET prepare-docs-site)
	add_custom_target(prev-clean-docs
		COMMAND "${CMAKE_COMMAND}" -E rm -rf "${SPHINX_DOCTREE_DIR}/weaver"
		COMMAND "${CMAKE_COMMAND}" -E make_directory "${SPHINX_DOCTREE_DIR}/weaver"
		COMMENT "Cleaning Sphinx doctrees while preserving the staged site..."
	)
	add_dependencies(prev-clean-docs prepare-docs-site)
else()
	add_custom_target(prev-clean-docs
		COMMAND "${CMAKE_COMMAND}" -E rm -rf "${SPHINX_DOCTREE_DIR}/weaver"
		COMMAND "${CMAKE_COMMAND}" -E make_directory "${SPHINX_DOCTREE_DIR}/weaver"
		COMMENT "Cleaning previous Sphinx build output..."
	)
endif()

add_custom_target(update-docs-catalog-weaver
	COMMAND "${CMAKE_COMMAND}" -E make_directory "${SPHINX_DOCS_OUTPUT_DIR}"
	COMMAND "${Python3_EXECUTABLE}" "${DOCS_DIR}/UpdateDocsCatalog.py"
		--catalog "${DOCS_WEAVER_CATALOG}"
		--version "${DOCS_WEAVER_VERSION}"
		--channel "${DOCS_WEAVER_CHANNEL}"
		--latest "$<IF:$<BOOL:${DOCS_WEAVER_LATEST}>,true,false>"
		--required-assets-dir "${PROJECT_SOURCE_DIR}/manager/releases"
	COMMENT "Updating Weaver documentation version catalog..."
)
add_dependencies(update-docs-catalog-weaver prev-clean-docs)

add_custom_target(build-docs-weaver)
set(WEAVER_DOCS_LANGUAGE_TARGETS)
set(WEAVER_DOCS_VERSION_OUTPUT_DIR "${SPHINX_DOCS_OUTPUT_DIR}/${DOCS_WEAVER_VERSION}")

foreach(language IN LISTS ALL_LANGUAGES)
	set(language_docs_output_dir "${WEAVER_DOCS_VERSION_OUTPUT_DIR}")
	if(NOT language STREQUAL DEFAULT_LANGUAGE)
		set(language_docs_output_dir "${language_docs_output_dir}/${language}")
	endif()

	set(language_target "build-doc-weaver-${language}")
	add_custom_target(${language_target}
		COMMAND "${CMAKE_COMMAND}" -E env
			"WEAVER_DOCS_VERSION=${DOCS_WEAVER_VERSION}"
			"WEAVER_DOCS_LANGUAGE=${language}"
			"WEAVER_DOCS_CATALOG=${DOCS_WEAVER_CATALOG}"
			"WEAVER_DOCS_SITE_ROOT=${SPHINX_DOCS_OUTPUT_DIR}"
			"${Python3_EXECUTABLE}" -m sphinx
			-E
			-c "${WEAVER_DOCS_SOURCE_DIR}/latest"
			-b html
			-d "${SPHINX_DOCTREE_DIR}/weaver/${DOCS_WEAVER_VERSION}-${language}"
			-q
			-j 1
			"${WEAVER_DOCS_SOURCE_DIR}/latest"
			"${language_docs_output_dir}"
		WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}"
		COMMENT "Building Weaver documentation ${DOCS_WEAVER_VERSION} (${language})..."
	)
	add_dependencies(${language_target} update-docs-catalog-weaver)
	if(TARGET build-doxygen)
		add_dependencies(${language_target} build-doxygen)
	endif()
	list(APPEND WEAVER_DOCS_LANGUAGE_TARGETS ${language_target})
endforeach()

add_dependencies(build-docs-weaver ${WEAVER_DOCS_LANGUAGE_TARGETS})
if(TARGET build-doxygen)
	add_dependencies(build-docs-weaver build-doxygen)
endif()
